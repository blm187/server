# 语音分离配置（音源分离）
# 推荐方案选择：
# 1. pyannote（推荐）- 易集成，支持多人场景（2-5人），社区活跃
# 2. mossformer2 - 性能优秀，需要SpeechBrain环境，适合2-3人场景
# 3. conv_tasnet - 速度快，资源占用低，主要适合2人场景
speech_separation:
  enabled: false  # 是否启用语音分离（模型已就绪，已启用）
  provider: conv_tasnet  # 选择分离方案：conv_tasnet（完整5阶段Pipeline）
  
  conv_tasnet:
    # Conv-TasNet完整Pipeline配置（5阶段：分离→VAD→x-vector→聚类→时间戳）
    model_name: "models/ConvTasNet_Libri2Mix_sepnoisy_16k"  # 使用本地模型路径
    max_speakers: 2  # 最多说话人数量（对应模型支持的人数）

    use_webrtc_vad: true  # 是否使用WebRTC VAD3（阶段2）
    use_ecapa: true  # 是否使用ECAPA-TDNN提取x-vector（阶段3），需要speechbrain
    energy_threshold: 0.02  # 能量阈值（提高以过滤更多噪声，单人场景建议0.015-0.02），现在会使用此配置值
    skip_single_speaker: true  # 是否跳过单人场景（false=强制启用分离，true=单人场景跳过分离，推荐设为true）

    single_speaker_detection_method: "hybrid"  # 检测方法："simple"（简单）| "enhanced"（增强）| "xvector"（x-vector）| "hybrid"（混合，推荐）

    streaming_enabled: true  # 是否启用流式处理（true=流式模式，false=批量模式）

    overlap_size_ms: 50  # 边界平滑重叠大小（毫秒），用于消除边界噪声，推荐50ms
    # 流式处理增强配置
    agc_enabled: true  # 是否启用实时AGC（自动增益控制）
    agc_target_rms: 0.1  # AGC目标RMS（0.1推荐）
